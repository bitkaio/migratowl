  kubectl -n {{ .Release.Namespace }} port-forward svc/{{ include "migratowl.fullname" . }} 8000:{{ .Values.service.port }}
  curl localhost:8000/healthz
{{- if not .Values.existingSecret }}

WARNING: existingSecret is not set, so the server has no LLM API key and no MIGRATOWL_API_TOKEN.
  kubectl -n {{ .Release.Namespace }} create secret generic migratowl-env --from-literal=ANTHROPIC_API_KEY=... --from-literal=MIGRATOWL_API_TOKEN="$(openssl rand -hex 32)"
  helm upgrade {{ .Release.Name }} <chart> --reuse-values --set existingSecret=migratowl-env
{{- end }}
